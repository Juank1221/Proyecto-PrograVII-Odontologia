-- PROCEDIMIENTO ALMACENADO SP_LISTAR_CATEGORIAS

CREATE OR REPLACE PROCEDURE SP_LISTAR_CATEGORIAS (
    P_CURSOR OUT SYS_REFCURSOR
)
AS
BEGIN
    OPEN P_CURSOR FOR
        SELECT
            ID_CATEGORIA,
            NOMBRE,
            DESCRIPCION
        FROM CATEGORIAS
        ORDER BY ID_CATEGORIA;
END SP_LISTAR_CATEGORIAS;

-- PROCEDIMIENTO ALMACENADO SP_LISTAR_PRODUCTOS

CREATE OR REPLACE PROCEDURE sp_listar_productos (
    P_CURSOR OUT SYS_REFCURSOR
)
AS
BEGIN
    OPEN P_CURSOR FOR
        SELECT
            P.ID_PRODUCTO,
            P.ID_CATEGORIA,
            C.NOMBRE AS CATEGORIA,
            P.CODIGO,
            P.NOMBRE,
            P.DESCRIPCION,
            P.UNIDAD_MEDIDA,
            P.STOCK_MINIMO,
            P.PRECIO_COSTO,
            P.PRECIO_VENTA,
            P.REQUIERE_CADUCIDAD,
            P.ACTIVO
        FROM PRODUCTOS P
        INNER JOIN CATEGORIAS C
            ON P.ID_CATEGORIA = C.ID_CATEGORIA
        ORDER BY P.ID_PRODUCTO;
END sp_listar_productos;

-- PROCEDIMIENTO ALMACENADO SP_OBTENER_PRODUCTO

CREATE OR REPLACE PROCEDURE SP_OBTENER_PRODUCTO (
    P_ID_PRODUCTO IN PRODUCTOS.ID_PRODUCTO%TYPE,
    P_CURSOR OUT SYS_REFCURSOR
)
AS
BEGIN
    OPEN P_CURSOR FOR
        SELECT
            P.ID_PRODUCTO,
            P.ID_CATEGORIA,
            C.NOMBRE AS CATEGORIA,
            P.CODIGO,
            P.NOMBRE,
            P.DESCRIPCION,
            P.UNIDAD_MEDIDA,
            P.STOCK_MINIMO,
            P.PRECIO_COSTO,
            P.PRECIO_VENTA,
            P.REQUIERE_CADUCIDAD,
            P.ACTIVO
        FROM PRODUCTOS P
        INNER JOIN CATEGORIAS C
            ON P.ID_CATEGORIA = C.ID_CATEGORIA
        WHERE P.ID_PRODUCTO = P_ID_PRODUCTO;
END SP_OBTENER_PRODUCTO;

-- PROCEDIMIENTO ALMACENADO SP_INSERTAR_PRODUCTO

CREATE OR REPLACE PROCEDURE SP_INSERTAR_PRODUCTO (
    P_ID_CATEGORIA       IN PRODUCTOS.ID_CATEGORIA%TYPE,
    P_CODIGO             IN PRODUCTOS.CODIGO%TYPE,
    P_NOMBRE             IN PRODUCTOS.NOMBRE%TYPE,
    P_DESCRIPCION        IN PRODUCTOS.DESCRIPCION%TYPE,
    P_UNIDAD_MEDIDA      IN PRODUCTOS.UNIDAD_MEDIDA%TYPE,
    P_STOCK_MINIMO       IN PRODUCTOS.STOCK_MINIMO%TYPE,
    P_PRECIO_COSTO       IN PRODUCTOS.PRECIO_COSTO%TYPE,
    P_PRECIO_VENTA       IN PRODUCTOS.PRECIO_VENTA%TYPE,
    P_REQUIERE_CADUCIDAD IN PRODUCTOS.REQUIERE_CADUCIDAD%TYPE,
    P_ACTIVO             IN PRODUCTOS.ACTIVO%TYPE,
    P_ID_PRODUCTO        OUT PRODUCTOS.ID_PRODUCTO%TYPE
)
AS
BEGIN
    INSERT INTO PRODUCTOS (
        ID_CATEGORIA,
        CODIGO,
        NOMBRE,
        DESCRIPCION,
        UNIDAD_MEDIDA,
        STOCK_MINIMO,
        PRECIO_COSTO,
        PRECIO_VENTA,
        REQUIERE_CADUCIDAD,
        ACTIVO
    )
    VALUES (
        P_ID_CATEGORIA,
        P_CODIGO,
        P_NOMBRE,
        P_DESCRIPCION,
        P_UNIDAD_MEDIDA,
        P_STOCK_MINIMO,
        P_PRECIO_COSTO,
        P_PRECIO_VENTA,
        P_REQUIERE_CADUCIDAD,
        P_ACTIVO
    )
    RETURNING ID_PRODUCTO INTO P_ID_PRODUCTO;

END SP_INSERTAR_PRODUCTO;

-- PROCEDIMIENTO ALMACENADO SP_ACTUALIZAR_PRODUCTO

CREATE OR REPLACE PROCEDURE SP_ACTUALIZAR_PRODUCTO (
    P_ID_PRODUCTO        IN PRODUCTOS.ID_PRODUCTO%TYPE,
    P_ID_CATEGORIA       IN PRODUCTOS.ID_CATEGORIA%TYPE,
    P_CODIGO             IN PRODUCTOS.CODIGO%TYPE,
    P_NOMBRE             IN PRODUCTOS.NOMBRE%TYPE,
    P_DESCRIPCION        IN PRODUCTOS.DESCRIPCION%TYPE,
    P_UNIDAD_MEDIDA      IN PRODUCTOS.UNIDAD_MEDIDA%TYPE,
    P_STOCK_MINIMO       IN PRODUCTOS.STOCK_MINIMO%TYPE,
    P_PRECIO_COSTO       IN PRODUCTOS.PRECIO_COSTO%TYPE,
    P_PRECIO_VENTA       IN PRODUCTOS.PRECIO_VENTA%TYPE,
    P_REQUIERE_CADUCIDAD IN PRODUCTOS.REQUIERE_CADUCIDAD%TYPE,
    P_ACTIVO             IN PRODUCTOS.ACTIVO%TYPE
)
AS
BEGIN
    UPDATE PRODUCTOS
    SET
        ID_CATEGORIA       = P_ID_CATEGORIA,
        CODIGO             = P_CODIGO,
        NOMBRE             = P_NOMBRE,
        DESCRIPCION        = P_DESCRIPCION,
        UNIDAD_MEDIDA      = P_UNIDAD_MEDIDA,
        STOCK_MINIMO       = P_STOCK_MINIMO,
        PRECIO_COSTO       = P_PRECIO_COSTO,
        PRECIO_VENTA       = P_PRECIO_VENTA,
        REQUIERE_CADUCIDAD = P_REQUIERE_CADUCIDAD,
        ACTIVO             = P_ACTIVO
    WHERE ID_PRODUCTO = P_ID_PRODUCTO;

    IF SQL%ROWCOUNT = 0 THEN
        RAISE_APPLICATION_ERROR(
            -20001,
            'No se encontró el producto indicado.'
        );
    END IF;

END SP_ACTUALIZAR_PRODUCTO;

-- PROCEDIMIENTO ALMACENADO DESACTIVAR PRODUCTO

CREATE OR REPLACE PROCEDURE SP_DESACTIVAR_PRODUCTO (
    P_ID_PRODUCTO IN PRODUCTOS.ID_PRODUCTO%TYPE
)
AS
BEGIN
    UPDATE PRODUCTOS
    SET ACTIVO = 0
    WHERE ID_PRODUCTO = P_ID_PRODUCTO;

    IF SQL%ROWCOUNT = 0 THEN
        RAISE_APPLICATION_ERROR(
            -20001,
            'No se encontró el producto indicado.'
        );
    END IF;

END SP_DESACTIVAR_PRODUCTO;

-- PROCEDIMIENTO ALMACENADO ACTIVAR PRODUCTO

CREATE OR REPLACE PROCEDURE SP_ACTIVAR_PRODUCTO (
    P_ID_PRODUCTO IN PRODUCTOS.ID_PRODUCTO%TYPE
)
AS
BEGIN
    UPDATE PRODUCTOS
    SET ACTIVO = 1
    WHERE ID_PRODUCTO = P_ID_PRODUCTO;

    IF SQL%ROWCOUNT = 0 THEN
        RAISE_APPLICATION_ERROR(
            -20001,
            'No se encontró el producto indicado.'
        );
    END IF;
END SP_ACTIVAR_PRODUCTO;
